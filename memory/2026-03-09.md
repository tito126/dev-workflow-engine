```python
def extract_error_reason(content: str) -> str:
    """提取错误的真正原因"""
    # 尝试提取各种格式的错误原因
    patterns = [
        r'原因[：:]\s*(.+?)(?:[,，。]|$)',

        r'错误[：:]\s*(.+?)(?:[,，。]|$)',
        # ...
    ]
    # 返回提取的原因

def categorize_error(content: str) -> dict:
    # 1. 先提取真正的错误原因
    error_reason = extract_error_reason(content)
    
    # 2. 业务错误优先匹配（最高优先级）
    business_keywords = [
        '批次冻结失败', '库存不足', '标签算法', 
        'buildData失败', '消息处理事件'
    ]
    
    # 3. 返回详细信息
    return {
        'category': 'business_error',
        'name': '业务逻辑错误',
        'error_reason': error_reason,  # 真正的原因
        'suggestion': '...'
    }
```
